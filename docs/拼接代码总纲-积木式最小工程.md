## 二、Rust 控制面（`crates/` 每 crate 一类作用）

### A. 子进程监督 —— 从 OpenWorker/AgentForge 拼接（作用：只负责拉起/回收 Python sidecar，不写业务）

```rust
// crates/process-supervisor/src/supervisor.rs —— 一个文件=一个作用：监督子进程生命周期
use std::path::PathBuf;
use std::process::{Child, Command};
use std::sync::Mutex;

pub struct SidecarSupervisor { child: Mutex<Option<Child>> }

impl SidecarSupervisor {
    pub fn new() -> Self {
        Self { child: Mutex::new(None) }
    }

    /// 拉起 Python sidecar：注入 launch token + 端口；随父进程自退（OpenWorker 式）
    pub fn spawn(&self, server_bin: PathBuf, port: u16, token: &str) -> std::io::Result<()> {
        let mut cmd = Command::new(server_bin);
        cmd.args(["--host", "127.0.0.1", "--port", &port.to_string()]);
        // OpenWorker: COWORKER_EXIT_WITH_PARENT=1 让 sidecar 检测到父进程退出即自杀
        cmd.env("EXIT_WITH_PARENT", "1")
           .env("PARENT_PID", std::process::id().to_string())
           .env("SIDECAR_TOKEN", token);          // 一次性 token
        #[cfg(windows)] { use std::os::windows::process::CommandExt;
            cmd.creation_flags(0x08000000); }      // CREATE_NO_WINDOW
        let child = cmd.spawn()?;
        *self.child.lock().unwrap() = Some(child);
        Ok(())
    }

    /// 退出时回收：Windows 用 taskkill（AgentForge 式），Unix 直接 kill
    pub fn shutdown(&self) {
        if let Some(mut c) = self.child.lock().unwrap().take() {
            #[cfg(target_os = "windows")]
            let _ = Command::new("taskkill").args(["/F", "/T", "/PID", &c.id().to_string()]).spawn();
            let _ = c.kill();
        }
    }
}
```

### B. 凭证 Vault —— 作用：只管密钥存取，不碰业务（倒 用常量时间比较）

```rust
// crates/security-vault/src/vault.rs —— 一个文件=一个作用：本地凭证存取
use std::collections::HashMap;
use std::sync::Mutex;

pub struct Vault(Mutex<HashMap<String, String>>);

impl Vault {
    pub fn new() -> Self { Self(Mutex::new(HashMap::new())) }
    pub fn set(&self, k: String, v: String) { self.0.lock().unwrap().insert(k, v); }
    /// 只允许返回短生命周期 capability token 的持有者取用；不回显
    pub fn get(&self, k: &str) -> Option<String> { self.0.lock().unwrap().get(k).cloned() }
}

/// 常量时间比较 token（OpenWorker 用 secrets.compare_digest；Rust 用强化比较避免时序侧信道）
pub fn constant_time_eq(a: &str, b: &str) -> bool {
    if a.len() != b.len() { return false; }
    a.bytes().zip(b.bytes()).fold(0u8, |acc, (x, y)| acc | (x ^ y)) == 0
}
```

## 三、TypeScript Agent 编排（`packages/` 每 package 一类作用）

### D. 最小 DAG 执行器 —— 从 DeepSeek Harness/CoreCoder 思路纯化（作用：只做「规划→执行→事件」语义）

```ts
// packages/agent-runtime/src/executor.ts —— 一个文件=一个作用：DAG 执行
import type { PlanStep, TaskEvent } from '@app/protocol';

export type Emit = (e: TaskEvent) => void;

/** DAG 节点：每节点有输入/输出/预算/幂等键（Node 即一种作用） */
export interface DAGNode {
  id: string;
  kind: 'model' | 'tool';
  budget?: number;
  idempotencyKey?: string;
  deps: string[];
}

export class DAGExecutor {
  constructor(private emit: Emit) {}

  async run(nodes: DAGNode[]): Promise<void> {
    const pending = new Map(nodes.map(n => [n.id, n]));
    while (pending.size) {
      const ready = [...pending.values()].filter(n => n.deps.every(d => !pending.has(d)));
      for (const node of ready) {
        this.emit({ type: 'tool.called', taskId: 'T1', callId: node.id,
                    inputHash: node.idemacyKey ?? node.id, at: Date.now() });
        // 由 capability runtime 执行；此处不写真实工具，只发事件
        this.emit({ type: 'tool.result', taskId: 'T1', callId: node.id,
                    status: 'ok', outputRef: `ref:${node.id}`, at: Date.now() });
        pending.delete(node.id);
      }
      if (!ready.length) break; // 防环：DAG 不允许成环（构造期校验）
    }
  }
}
```

export interface ChatRequest { model: string; messages: { role: string; content: string }[]; stream: boolean }
export interface ModelDriver {

  id(): string;            // 'openai' | 'anthropic' | 'local' | ...

// packages/provider-sdk/src/adapters/openai.ts —— 一个文件=一个作用：OpenAI-compatible 适配
import type { ModelDriver, ChatRequest } from '../driver';
export class OpenAICompatible implements ModelDriver {
  constructor(private baseUrl: string) {}
  id() { return 'openai' as string; }
  async *chat(req: ChatRequest, apiKey: string): AsyncIterable<string> {
    // SSE 流式解析（AgentForge 手写 fetch+SSE 的同款）
    const r = await fetch(`${this.baseUrl}/v1/chat/completions`, {
      method: 'POST',
      headers: { 'content-type': 'application/json', 'authorization': `Bearer ${apiKey}` },
      body: JSON.stringify({ ...req, stream: true }),
    });
    const reader = r.body!.getReader(); const dec = new TextDecoder(); let buf = '';
    while (true) { const { done, value } = await reader.read(); if (done) break;
      buf += dec.decode(value, { stream: true });
      for (const line of buf.split('\n')) { if (!line.startsWith('data:')) continue;
        const d = line.slice(5).trim(); if (d === '[DONE]') return;
        try { yield JSON.parse(d).choices?.[0]?.delta?.content ?? ''; } catch { /* partial */ } }
    }
  }
}

// packages/provider-sdk/src/router.ts —— 一个文件=一个作用：按任务选模型
import type { ModelDriver } from './driver';
export class ModelRouter {
  constructor(private drivers: Map<string, ModelDriver>) {}

## 四、Python 重计算 sidecar（FastAPI + token，127.0.0.1）

### F. FastAPI 骨架 —— 从 OpenWorker 拼接（作用：只建路由+鉴权，具体处理交给 processor）

```python
# sidecars/document-worker/app.py —— 一个文件=一个作用：路由+鉴权中间件（不开业务处理）
import os, secrets
from fastapi import FastAPI, Header, WebSocket, HTTPException
from .processor import process_document   # port: 只依赖文档处理接口

app = FastAPI()
SIDECAR_TOKEN = os.environ.get("SIDECAR_TOKEN", "")

def check_token(x_token: str | None) -> bool:
    return bool(x_token) and secrets.compare_digest(x_token, SIDECAR_TOKEN)  # OpenWorker 式

@app.post("/v1/doc/parse")
async def parse(body: dict, x_openworker_token: str | None = Header(default=None)):
    if not check_token(x_openworker_token):
        raise HTTPException(status_code=401, detail="bad token")
    return await process_document(body["inputRef"])   # 只调 processor，不开业务

@app.websocket("/ws/events")
async def ws_events(ws: WebSocket):
    subs = {p.strip() for p in ws.headers.get("sec-websocket-protocol", "").split(",") if p.strip()}
    if not any(secrets.compare_digest(s, SIDECAR_TOKEN) for s in subs):
        await ws.close(code=4401); return
    await ws.accept(subprotocol=SIDECAR_TOKEN)
    while True:  # 事件转发（OpenWorker 模式）
        msg = await ws.receive_json(); await ws.send_json({"echo": msg})
```

## 五、三语言整体接线（一个文件=一个作用，靠通道对齐）

| 通道 | 谁→谁 | 用什么 | 对应代码 |
|---|---|---|---|
| C1 invoke | React↔Rust | Tauri | `Rust SidecarSupervisor.spawn()` + 前端 `invoke('start')` |
| C2 HTTP/WS | TS/前端↔Python | `127.0.0.1:port` + token | `backendFetch`(OpenWorker helpers 式) + F 的 `check_token` |
| C3 JSON-RPC | TS→Rust | 事件上报 | `DAGExecutor.emit`(Emit 事件) |
| C4 port | TS 模块间 | 接口 | `ModelDriver` / `DAGNode` |
| C6 EventStream | UI↔领域 | 订阅 | `packages/protocol` `TaskEvent` |

**接线铁律**：UI 组件只 `emit('tool.called')`；Rust 只 `supervisor.spawn()`/`vault.get()`；Python 只 `process_document()`。任何文件不写第二种作用的逻辑。
